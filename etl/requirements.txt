# Relocation Finder ETL — Phase 1
#
# Deliberately minimal. No geopandas: the only "spatial" operation in Phase 1 is
# nearest-station-to-county-centroid, which is haversine math on lat/lon pairs.
# Census publishes county centroids directly (Gazetteer files), so we never need
# polygon geometry at this stage. Revisit when tract boundaries arrive (Phase 5+).

# Developed and offline-tested against pandas 3.0.x. The 2.x floor is kept
# because nothing here needs 3.x-only behaviour; 4.x is excluded until tested.
pandas>=2.0,<4
requests>=2.28
numpy>=1.24

# Only needed to read the Census CBSA delineation file, which ships as .xlsx.
# If this is a problem in your environment, see sources/bea.py — there is a
# documented CSV fallback path.
openpyxl>=3.1

# Optional, for .env loading. The code works without it if you export vars yourself.
python-dotenv>=1.0
