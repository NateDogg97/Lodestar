# Relocation Finder ETL — Phase 1
#
# Deliberately minimal. No geopandas: the county pipeline only does haversine
# math on lat/lon pairs. Phase 8 (tracts) needs point-in-polygon — which school
# district, neighborhood or police jurisdiction a tract falls in — so shapely
# was added then. Shapefiles are converted to GeoJSON with mapshaper (npx).

# Developed and offline-tested against pandas 3.0.x. The 2.x floor is kept
# because nothing here needs 3.x-only behaviour; 4.x is excluded until tested.
pandas>=2.0,<4
requests>=2.28
numpy>=1.24

# Only needed to read the Census CBSA delineation file, which ships as .xlsx.
# If this is a problem in your environment, see sources/bea.py — there is a
# documented CSV fallback path.
openpyxl>=3.1

# Optional, for .env loading. The code works without it if you export vars yourself.
python-dotenv>=1.0

# Phase 8 (etl/tracts/): point-in-polygon lookups.
shapely>=2.0,<3
# Phase 8: reads EPA's walkability geodatabase (attributes only; bundles GDAL).
pyogrio>=0.10
# Phase 8c: uploads tract files to Cloudflare R2 (S3-compatible API; tracts/upload.py).
boto3>=1.34
